使用例

TObservable<f32> hp { 100.0f };
TObservable<f32> max_hp { 100.0f };
TDerived<f32, f32> ratio(
    [](const f32& h, const f32& m){ return h / (m > 0 ? m : 1.0f); },
    hp, max_hp);
ratio.Get();      // 1.0
hp.Set(50.0f);   // 依存通知中に ratio を 0.5 へ即時再計算
ratio.Get();      // 0.5
