使用例

class FPlayerViewModel : public CViewModel {
public:
    TObservable<f32>  hp    { 100.0f };
    TObservable<i32>  level { 1 };
    TObservable<bool> invincible { false };
};
FPlayerViewModel vm;
vm.hp.Set(vm.hp.Get() - 10.0f);   // View 側へ自動反映
