使用例

TObservableArray<int> inv;
inv.Subscribe([](EArrayChange kind, usize idx, const int* v, void*){
    if (kind == EArrayChange::Inserted) ACS_LOG_INFO("Add[%zu]=%d", idx, *v);
}, nullptr);
inv.Add(7);
inv.SetAt(0, 99);
inv.RemoveAt(0);
