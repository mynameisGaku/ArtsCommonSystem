使用例

TObservable<f32> hp { 100.0f };         // 初期値つき
auto h = hp.Subscribe([](const f32& v, void*){
    ACS_LOG_INFO("HP=%.1f", v);              // 値が変わるたび呼ばれる
}, nullptr);
hp.Set(75.0f);                              // → 通知発火
hp.Unsubscribe(h);                          // 監視解除
