使用例

TMessagePipe<FDamageEvent> pipe;

// 生産スレッド:
pipe.Push(FDamageEvent{ enemy, 25.0f });

// 消費スレッド (毎フレーム):
FDamageEvent e;
while (pipe.TryPop(e)) {
    ApplyDamage(e);   // 溜まっている分を全部処理
}
