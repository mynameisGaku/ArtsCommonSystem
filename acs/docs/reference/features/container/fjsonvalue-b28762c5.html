使用例

auto r = acs::ParseJson(text);
if (r.IsOk()) {
    const FJsonValue& root = r.Value();
    FStringView img = root.Get("meta").Get("image").AsString();
    u32 n = root.Get("frames").Size();
    for (u32 i = 0; i < n; ++i)
        i32 x = root.Get("frames").At(i).Get("x").AsInt();
}
