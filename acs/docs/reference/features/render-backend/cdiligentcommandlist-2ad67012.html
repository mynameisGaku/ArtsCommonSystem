使用例

auto cmd = CreateRhiCommandList(*dev).Value();
cmd->Begin();
u32 i = swap->AcquireNextImage();
cmd->BeginRenderToSwapchain(*swap, i, {0,0,0,1});
cmd->SetPipeline(*pso);
cmd->SetConstantBuffer(0, *cb);
cmd->SetTexture(0, *tex);
cmd->DrawIndexed(idxCount);
cmd->EndRenderToSwapchain(*swap, i);
cmd->End();
if (!cmd->Submit() || !swap->Present()) { /* 描画ループを停止 */ }
