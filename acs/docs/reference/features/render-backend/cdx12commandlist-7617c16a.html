使用例

auto cmd = CreateRhiCommandList(*dev).Value();
cmd->Begin();
u32 i = swap->AcquireNextImage();
cmd->BeginRenderToSwapchain(*swap, i, {0.1f,0.1f,0.2f,1});
cmd->SetPipeline(*pso);
cmd->SetVertexBuffer(*vb, sizeof(FVertex));
cmd->Draw(3);
cmd->EndRenderToSwapchain(*swap, i);
cmd->End();
if (!cmd->Submit() || !swap->Present()) { /* 描画ループを停止 */ }
