使用例

class ARotateComponent : public AComponent {
public:
    ACS_GAME_COMPONENT_KIND(ARotateComponent)
    void OnUpdate(f32 dt) noexcept override {
        Owner().SetRotation2D(Owner().Rotation2D() + m_Speed * dt);
    }
    f32 m_Speed = 1.0f;
};
node->AddComponent<ARotateComponent>();
