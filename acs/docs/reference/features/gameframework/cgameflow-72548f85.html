使用例

CGameFlow flow;
flow.SetOnEnterCallback(EFlowState::Gameplay, &OnGameplayEnter, this);
flow.Init(EFlowState::Splash);
// 毎フレーム:
flow.Tick(dt);
if (pressed_start && flow.CurrentState() == EFlowState::MainTitle)
    flow.RequestTransition(EFlowState::MainMenu, 0.5f);
if (flow.IsTransitioning()) DrawFadeOverlay(flow.FadeProgress());
