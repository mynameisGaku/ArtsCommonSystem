CLlmSafetyPipeline

使用例

CLlmSafetyPipeline pipe;
pipe.Init();
pipe.SetTokenBudget(1024, 512);
pipe.SetCharacterAnchor("You are a friendly shopkeeper.");
auto in = pipe.ValidateInput(player_text);
if (in.verdict != ESafetyVerdict::Pass) { ShowRefusal(in.refusal_reason); return; }
auto out = pipe.FilterOutput(llm_text);
if (out.verdict == ESafetyVerdict::Refused) { NpcSilent(); return; }
Display(out.filtered_text);
