使用例

void OnRender(FRenderContext& rc) noexcept override {
    if (rc.HasSprites()) {
        rc.Sprites().DrawString(rc.GetFont(), "SCORE", x, y);
    }
    u32 w = rc.Width(), h = rc.Height();
}
