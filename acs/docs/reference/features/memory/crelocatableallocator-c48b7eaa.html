使用例

CRelocatableAllocator ra;
ra.Init(1u << 20, 4096);             // 1MB アリーナ / ハンドル最大 4096
FRelocHandle h = ra.Alloc(128);          // 失敗時は h.IsValid()==false
void* p = ra.Resolve(h);                 // 使う直前に解決(次の Compact まで有効)
ra.Compact();                            // 隙間を詰める→生ポインタは無効化、ハンドルは有効
p = ra.Resolve(h);                       // Compact 後は取り直す
