使用例

AModelInspectorPanel insp;
insp.Init();
workspace.RegisterPanel(&insp);

FMeshSummary    sum;       // 集計を埋める
FSubmeshInfo    subs[1]  = { { "Body", 0, 3600, 0 } };
insp.UpdateFromModel(sum, subs, 1,
                     nullptr, 0,    // bones なし
                     nullptr, 0);   // clips なし
