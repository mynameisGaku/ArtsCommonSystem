使用例

namespace ac = acs::game::animcurve;
ac::AAnimCurveEditorPanel panel;
panel.Init();
acs::game::FAnimationCurve curve;  // 別途 key 追加済とする
panel.SetCurve(&curve);
panel.SetOnChangeCallback([](void* user, acs::game::FAnimationCurve* c) noexcept {
    (void)user; (void)c;   // autosave 等
}, nullptr);
if (panel.IsDirty()) panel.ClearDirty();
panel.Shutdown();
