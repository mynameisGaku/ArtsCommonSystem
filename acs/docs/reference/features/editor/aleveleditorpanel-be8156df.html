ALevelEditorPanel

使用例

namespace lv = acs::game::leveledit;
lv::ALevelEditorPanel panel;
panel.Init();
workspace.RegisterPanel(&panel);

acs::game::FTilemap map;  // 別途 Init 済とする
panel.SetTilemap(&map);
panel.SetCurrentBrush(lv::EBrushKind::Paint);
panel.SetCurrentTileId(1);
// ... TickAllPanels(dt) 内で DrawUI ...
workspace.UnregisterPanel(&panel);
panel.Shutdown();
