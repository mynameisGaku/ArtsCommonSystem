使用例

CSpriteBatch sb;
sb.Init(*rdr.Device(), rdr.ColorFormat(), /*max_sprites=*/4096);
sb.Begin(*rdr.CommandList(), screen_w, screen_h);
sb.Draw(tex, 100, 200, 64, 64);                    // 64x64 を (100,200) に
sb.DrawRect(0, 0, screen_w, 32, FVec4{0,0,0,0.5f}); // 半透明バー
sb.DrawString(font, "スコア 1000", 16, 16, FVec4{1,1,1,1});
sb.End();
