使用例

CMotionVector mv;
mv.Init(*dev, w, h);
const u32 motion_draws = static_cast<u32>(visible_meshes.Size());
bool motion_valid = false;
if (mv.BeginFrame(motion_draws) &&
    mv.Begin(*cl, vp_no_jitter, prev_vp_no_jitter)) {
    bool complete = true;
    for (auto& m : visible_meshes)
        if (!mv.DrawMesh(*cl, m.gm, m.model, m.prev_model)) complete = false;
    mv.End(*cl);
    motion_valid = complete && mv.ObjectDrawCount() == motion_draws;
}
post_params.taa_motion_texture = motion_valid ? mv.OutputTexture() : nullptr;
if (motion_valid)
    ssr.Render(..., mv.OutputNormalTexture(), ..., mv.OutputTexture());
