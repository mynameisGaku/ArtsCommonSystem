使用例

FPipelineDesc pd;
pd.vs = vs.Get(); pd.ps = ps.Get();
pd.rt_format = EFormat::B8G8R8A8_UNorm;
pd.depth_format = EFormat::D32_Float;
pd.depth_test = true;
pd.cull_mode = ECullMode::Back;
pd.layout[0] = { "POSITION", 0, EFormat::R32G32B32_Float, 0 };
pd.layout_count = 1; pd.vertex_stride = 12;
auto pipe = CreateRhiPipeline(*dev, pd).Value();
