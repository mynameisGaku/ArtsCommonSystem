使用例

CPostProcess pp;
pp.Init(*rdr.Device(), w, h, rdr.ColorFormat());
// OnCustomFrame 内:
cl->BeginRenderToTexture(*pp.HdrRenderTarget(), clear, rdr.DepthBuffer());
// ... シーンを HDR で描画 ...
cl->EndRenderToTexture(*pp.HdrRenderTarget());
FPostProcessParams p; p.bloom_intensity = 0.6f; p.exposure = 1.0f;
pp.Render(*cl, *rdr.Swapchain(), buf_idx, p);
