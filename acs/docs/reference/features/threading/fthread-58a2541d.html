使用例

void Work(void* user) { /* 別スレッドで走る */ }
auto r = FThread::Spawn(&Work, &ctx);
if (r.IsOk()) {
    FThread t = static_cast<FThread&&>(r.Value());
    t.Join();   // 終わるまで待つ
}
